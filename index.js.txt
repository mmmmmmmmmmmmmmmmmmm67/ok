const { Client, GatewayIntentBits } = require('discord.js');
const { joinVoiceChannel } = require('@discordjs/voice');

const client = new Client({
    intents: [
        GatewayIntentBits.Guilds,
        GatewayIntentBits.GuildVoiceStates
    ]
});

client.once('ready', () => {
    console.log(`Logged in successfully as ${client.user.tag}!`);

    // هنا يحط الـ ID حق الروم الصوتي اللي بياخده من إعدادات Render
    const channelId = process.env.VOICE_CHANNEL_ID;
    const guildId = process.env.GUILD_ID; // آيدي السيرفر

    if (!channelId || !guildId) {
        console.log("Voice channel ID or Guild ID is missing in environment variables!");
        return;
    }

    const guild = client.guilds.cache.get(guildId);
    if (!guild) return;

    // يدخل الروم الصوتي
    joinVoiceChannel({
        channelId: channelId,
        guildId: guild.id,
        adapterCreator: guild.voiceAdapterCreator,
    });

    console.log("Successfully joined the voice channel!");
});

client.login(process.env.TOKEN);
